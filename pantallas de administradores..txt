  static void _mostrarCerrarSesion(
    BuildContext context,
  ) {
    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: const Text('Cerrar sesión'),
          content: const Text(
            '¿Estás seguro de que deseas cerrar sesión?',
          ),
          actions: [
            TextButton(
              onPressed: () {
                Navigator.pop(context);
              },
              child: const Text('Cancelar'),
            ),

            ElevatedButton(
              onPressed: () {
                Navigator.pop(context);

                // Próximamente:
                // regresar a la pantalla de inicio de sesión.
              },
              child: const Text('Cerrar sesión'),
            ),
          ],
        );
      },
    );
  }

    final texto = buscadorController.text.toLowerCase();

    return reportes.where((reporte) {
      final coincideTexto =
          reporte['folio'].toString().toLowerCase().contains(texto) ||
          reporte['ubicacion'].toString().toLowerCase().contains(texto);

      final coincideEstado = filtroEstado == 'Todos' ||
          reporte['estado'] == filtroEstado;

      final coincideTipo =
          filtroTipo == 'Todos' || reporte['tipo'] == filtroTipo;

      return coincideTexto && coincideEstado && coincideTipo;
    }).toList();

        return Colors.orange;

      case 'En atención':
        return Colors.deepOrange;

      case 'Atendido':
        return Colors.green;

      case 'Cerrado':
        return Colors.grey;

        return Colors.grey;

class _FiltroDropdown extends StatelessWidget {
  final String titulo;
  final String valor;
  final List<String> opciones;
  final ValueChanged<String> onChanged;

  const _FiltroDropdown({
    required this.titulo,
    required this.valor,
    required this.opciones,
    required this.onChanged,
  });

  @override
  Widget build(BuildContext context) {
    return Container(
      padding:
          const EdgeInsets.symmetric(horizontal: 12),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(14),
      ),
      child: DropdownButtonHideUnderline(
        child: DropdownButton<String>(
          value: valor,
          isExpanded: true,
          items: opciones.map((opcion) {
            return DropdownMenuItem(
              value: opcion,
              child: Text(
                opcion,
                overflow: TextOverflow.ellipsis,
              ),
            );
          }).toList(),
          onChanged: (valor) {
            if (valor != null) {
              onChanged(valor);
            }
          },
        ),
      ),
    );
  }
}

    switch (estado) {
      case 'Nuevo':
        return Colors.blue;

      case 'En revisión':
        return Colors.orange;

      case 'En atención':
        return Colors.deepOrange;

      case 'Atendido':
        return Colors.green;

      case 'Cerrado':

        return Colors.grey;

    });

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(
          'Estado cambiado a "$nuevoEstado"',
        ),
        duration: const Duration(seconds: 2),
      ),
    );
  }

  // GUARDAR COMENTARIO
  // =====================================================

    if (comentarioController.text.trim().isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Escribe un comentario antes de guardar.',
          ),
        ),
      );
      return;
    }

    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(
        content: Text(
          'Comentario agregado correctamente.',
        ),
      ),
    );

    comentarioController.clear();

                );

          ],
        );
      },

class _GraficaMensual extends StatelessWidget {
  final Map<String, int> datos;

  const _GraficaMensual({
    required this.datos,
  });

  @override
  Widget build(BuildContext context) {
    final maximo = datos.values.isEmpty
        ? 1
        : datos.values.reduce(
            (a, b) => a > b ? a : b,
          );

    return Row(
      crossAxisAlignment: CrossAxisAlignment.end,
      children: datos.entries.map((entrada) {
        final altura = maximo == 0
            ? 0.0
            : (entrada.value / maximo) * 145;

        return Expanded(
          child: Padding(
            padding: const EdgeInsets.symmetric(horizontal: 5),
            child: Column(
              mainAxisAlignment: MainAxisAlignment.end,
              children: [
                Text(
                  '${entrada.value}',
                  style: const TextStyle(
                    fontSize: 11,
                    fontWeight: FontWeight.bold,
                    color: Color(0xFF263238),
                  ),
                ),

                const SizedBox(height: 5),

                Container(
                  height: altura,
                  decoration: BoxDecoration(
                    color: const Color(0xFF2E7D32),
                    borderRadius: const BorderRadius.vertical(
                      top: Radius.circular(8),
                    ),
                  ),
                ),

                const SizedBox(height: 7),

                Text(
                  entrada.key,
                  style: const TextStyle(
                    fontSize: 11,
                    color: Colors.black54,
                  ),
                ),
              ],
            ),
          ),
        );
      }).toList(),
    );
  }
}

class _FiltroDropdown extends StatelessWidget {
  final String titulo;
  final String valor;
  final List<String> opciones;
  final ValueChanged<String> onChanged;

  const _FiltroDropdown({
    required this.titulo,
    required this.valor,
    required this.opciones,
    required this.onChanged,
  });

  @override
  Widget build(BuildContext context) {
    return Container(
      padding:
          const EdgeInsets.symmetric(horizontal: 12),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(14),
      ),
      child: DropdownButtonHideUnderline(
        child: DropdownButton<String>(
          value: valor,
          isExpanded: true,
          items: opciones.map((opcion) {
            return DropdownMenuItem(
              value: opcion,
              child: Text(
                opcion,
                overflow: TextOverflow.ellipsis,
              ),
            );
          }).toList(),
          onChanged: (valor) {
            if (valor != null) {
              onChanged(valor);
            }
          },
        ),
      ),
    );
  }
}

  Widget _botonVista({
    required IconData icono,
    required String texto,
    required bool seleccionado,
    required VoidCallback onTap,
  }) {
    return GestureDetector(
      onTap: onTap,
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 200),
        decoration: BoxDecoration(
          color: seleccionado
              ? const Color(0xFF1B5E20)
              : Colors.transparent,
          borderRadius: BorderRadius.circular(11),
        ),
        child: Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Icon(
              icono,
              size: 18,
              color: seleccionado
                  ? Colors.white
                  : Colors.grey.shade700,
            ),

            const SizedBox(width: 7),

            Text(
              texto,
              style: TextStyle(
                fontWeight: FontWeight.w600,
                fontSize: 13,
                color: seleccionado
                    ? Colors.white
                    : Colors.grey.shade700,
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _zonaCalor({
    double? left,
    double? right,
    double? top,
    double? bottom,
    required double size,
    required Color color,
    required double intensidad,
  }) {
    return Positioned(
      left: left,
      right: right,
      top: top,
      bottom: bottom,
      child: ClipOval(
        child: BackdropFilter(
          filter: ImageFilter.blur(
            sigmaX: 12,
            sigmaY: 12,
          ),
          child: Container(
            width: size,
            height: size,
            decoration: BoxDecoration(
              shape: BoxShape.circle,
              color: color.withValues(
                alpha: intensidad,
              ),
            ),
          ),
        ),
      ),
    );
  }

  Widget _marcadorMapa({
    double? left,
    double? right,
    required double top,
    required Color color,
    required Map<String, dynamic> reporte,
  }) {
    return Positioned(
      left: left,
      right: right,
      top: top,
      child: GestureDetector(
        onTap: () {
          mostrarDetalle(reporte);
        },
        child: Container(
          width: 42,
          height: 42,
          decoration: BoxDecoration(
            color: color,
            shape: BoxShape.circle,
            border: Border.all(
              color: Colors.white,

  Widget _filtro(String nombre) {
    final seleccionado = filtro == nombre;

  Future<List<String>> _seleccionarImagenes() async {
    final resultado = await FilePicker.platform.pickFiles(
      type: FileType.image,
      allowMultiple: true,
    );

    if (resultado == null) {
      return [];
    }

    return resultado.paths
        .whereType<String>()
        .toList();
  }

  // ============================================================
  // SELECCIONAR VIDEO
  // ============================================================

  Future<String?> _seleccionarVideo() async {
    final resultado = await FilePicker.platform.pickFiles(
      type: FileType.video,
      allowMultiple: false,
    );

    if (resultado == null || resultado.files.single.path == null) {
      return null;
    }

    return resultado.files.single.path;
  }
