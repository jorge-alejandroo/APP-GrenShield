  // ============================================================
  //
  // Revoca el JWT actual en FastAPI y regresa al login.
  // Si no hay conexión, elimina el token local y su caducidad
  // en el servidor ocurrirá según la configuración de JWT.
  // ============================================================

  Future<void> _cerrarSesion(BuildContext context) async {
    final confirmar = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Cerrar sesión'),
        content: const Text('¿Deseas cerrar tu sesión de GreenShield?'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('Cancelar')),
          FilledButton(onPressed: () => Navigator.pop(dialogContext, true),
            child: const Text('Cerrar sesión')),
        ],
      ),
    );
    if (confirmar != true) return;
    try {
      await GreenShieldApi.instance.logout();
    } catch (_) {
      // El JWT local se elimina incluso si no hay red; en el servidor
      // seguirá siendo válido hasta caducar si no pudo revocarse.
    }
    if (!context.mounted) return;
    Navigator.pushAndRemoveUntil(context,
      MaterialPageRoute(builder: (_) => const LoginScreen()), (_) => false);

  // ============================================================
  // CAMBIAR CORREO VERIFICADO / EDITAR TELÉFONO SIMULADO
  // ============================================================

  Future<void> _cambiarCorreo() async {
    final nuevoCorreo = await Navigator.push<String>(
      context,
      MaterialPageRoute(builder: (context) => const ChangeEmailScreen()),
    );
    if (!mounted || nuevoCorreo == null) return;
    setState(() => correo = nuevoCorreo);
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Correo electrónico actualizado correctamente.')),
    );
  }

  Future<void> _editarTelefono() async {
    final controller = TextEditingController(text: telefono);
    bool guardando = false;
    String? mensaje;

    await showDialog<void>(
      context: context,
      builder: (dialogContext) => StatefulBuilder(
        builder: (dialogBuildContext, actualizarDialogo) => AlertDialog(
          title: const Text('Cambiar Teléfono'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              TextField(
                controller: controller,
                keyboardType: TextInputType.phone,
                enabled: !guardando,
                decoration: const InputDecoration(
                  labelText: 'Teléfono',
                  border: OutlineInputBorder(),
                ),
              ),
              if (mensaje != null) ...[
                const SizedBox(height: 12),
                Text(
                  mensaje!,
                  style: const TextStyle(color: Colors.redAccent, fontSize: 13),
                ),
              ],
            ],
          ),
          actions: [
            TextButton(
              onPressed: guardando
                  ? null
                  : () => Navigator.of(dialogContext).pop(),
              child: const Text('Cancelar'),
            ),
            ElevatedButton(
              onPressed: guardando ? null : () async {
                final nuevoNumero = controller.text.trim();
                final soloDigitos = nuevoNumero.replaceAll(RegExp(r'[^0-9]'), '');
                if (soloDigitos.length < 10 || soloDigitos.length > 15) {
                  actualizarDialogo(() {
                    mensaje = 'Introduce un número válido de 10 a 15 dígitos.';
                  });
                  return;
                }
                if (nuevoNumero == telefono) {
                  Navigator.of(dialogContext).pop();
                  return;
                }
                final guardar = widget.guardarTelefonoEnApi;
                if (guardar == null) {
                  actualizarDialogo(() {
                    mensaje = 'No se guardó el teléfono: todavía falta conectar '
                        'el servicio de actualización del perfil con la API.';
                  });
                  return;
                }

                actualizarDialogo(() {
                  guardando = true;
                  mensaje = null;
                });
                try {
                  await guardar(nuevoNumero);
                  if (!mounted || !dialogContext.mounted) return;
                  setState(() => telefono = nuevoNumero);
                  Navigator.of(dialogContext).pop();
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(
                      content: Text('Teléfono actualizado correctamente.'),
                    ),
                  );
                } catch (error) {
                  if (dialogContext.mounted) {
                    actualizarDialogo(() {
                      mensaje = 'No se pudo actualizar el teléfono: $error';
                    });
                  }
                } finally {
                  if (dialogContext.mounted) {
                    actualizarDialogo(() => guardando = false);
                  }
                }
              },
              child: guardando
                  ? const SizedBox(
                      height: 18,
                      width: 18,
                      child: CircularProgressIndicator(strokeWidth: 2),
                    )
                  : const Text('Guardar'),
            ),
          ],
        ),
      ),
    );
    controller.dispose();
  }

  // ============================================================
  // CAMBIAR CONTRASEÑA
  // ============================================================

  void _cambiarPassword(BuildContext context) {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(
        content: Text(
          'La función para cambiar la contraseña se conectará posteriormente.',
        ),
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:latlong2/latlong.dart';

import '../widgets/report_location_picker.dart';

class FireReportScreen extends StatefulWidget {
  const FireReportScreen({super.key});

  @override
  State<FireReportScreen> createState() => _FireReportScreenState();
}

class _FireReportScreenState extends State<FireReportScreen> {
  // ============================================================
  // PASOS DEL REPORTE
  // ============================================================
  //
  // 0 = Información del incendio
  // 1 = Evidencia
  // 2 = Ubicación
  // 3 = Confirmación
  //
  // Más adelante podemos agregar o quitar pasos si el proyecto
  // cambia.
  // ============================================================

  int _currentStep = 0;

  // ============================================================
  // RESPUESTAS DEL CUESTIONARIO
  // ============================================================

  String? _tipoIncendio;
  String? _estadoIncendio;
  String? _queObserva;
  String? _riesgo;

  // Descripción escrita por el usuario.
  final TextEditingController _descripcionController =
      TextEditingController();

  // ============================================================
  // EVIDENCIAS
  // ============================================================
  //
  // Por ahora solamente guardamos indicadores para representar
  // las evidencias agregadas.
  //
  // Más adelante estos valores serán reemplazados por archivos
  // reales obtenidos mediante cámara, galería y grabadora.
  // ============================================================

  bool _tieneFoto = false;
  bool _tieneVideo = false;
  bool _tieneAudio = false;

  // ============================================================
  // UBICACIÓN
  // ============================================================

  bool _ubicacionConfirmada = false;
  LatLng? _coordenadasConfirmadas;

  // ============================================================
  // CONTROLADORES
  // ============================================================

  final PageController _pageController = PageController();

  @override
  void dispose() {
    _descripcionController.dispose();
    _pageController.dispose();
    super.dispose();
  }

  // ============================================================
  // AVANZAR AL SIGUIENTE PASO
  // ============================================================

  void _siguientePaso() {
    // Validamos cada sección antes de continuar.

    if (_currentStep == 0) {
      if (!_validarInformacion()) {
        return;
      }
    }

    if (_currentStep == 1) {
      if (!_validarEvidencia()) {
        return;
      }
    }

    if (_currentStep == 2) {
      if (!_ubicacionConfirmada) {
        _mostrarMensaje(
          'Confirma la ubicación del reporte para continuar.',
        );
        return;
      }
    }

    // Si todavía no estamos en el último paso,
    // avanzamos.
    if (_currentStep < 3) {
      setState(() {
        _currentStep++;
      });

      _pageController.animateToPage(
        _currentStep,
        duration: const Duration(milliseconds: 300),
        curve: Curves.easeInOut,
      );
    }
  }

  // ============================================================
  // REGRESAR AL PASO ANTERIOR
  // ============================================================

  void _pasoAnterior() {
    if (_currentStep > 0) {
      setState(() {
        _currentStep--;
      });

      _pageController.animateToPage(
        _currentStep,
        duration: const Duration(milliseconds: 300),
        curve: Curves.easeInOut,
      );
    }
  }

  // ============================================================
  // VALIDAR INFORMACIÓN
  // ============================================================

  bool _validarInformacion() {
    if (_tipoIncendio == null) {
      _mostrarMensaje(
        'Selecciona el tipo de incendio.',
      );
      return false;
    }

    if (_estadoIncendio == null) {
      _mostrarMensaje(
        'Indica si el incendio está activo.',
      );
      return false;
    }

    if (_queObserva == null) {
      _mostrarMensaje(
        'Indica qué observas.',
      );
      return false;
    }

    if (_riesgo == null) {
      _mostrarMensaje(
        'Indica si existe algún riesgo.',
      );
      return false;
    }

    return true;
  }

  // ============================================================
  // VALIDAR EVIDENCIA
  // ============================================================

  bool _validarEvidencia() {
    // Para que el reporte tenga evidencia, solicitamos al menos
    // una fotografía, video o audio.
    if (!_tieneFoto && !_tieneVideo && !_tieneAudio) {
      _mostrarMensaje(
        'Agrega al menos una evidencia: foto, video o audio.',
      );
      return false;
    }

    return true;
  }

  // ============================================================
  // MENSAJE TEMPORAL
  // ============================================================

  void _mostrarMensaje(String mensaje) {
    ScaffoldMessenger.of(context).hideCurrentSnackBar();

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(mensaje),
        behavior: SnackBarBehavior.floating,
      ),
    );
  }

  // ============================================================

  // PASOS DEL REPORTE
  // ============================================================
  //
  // 0 = Información
  // 1 = Evidencia
  // 2 = Ubicación
  // 3 = Confirmación
  // ============================================================

  int _currentStep = 0;

  // ============================================================
  // RESPUESTAS DEL CUESTIONARIO
  // ============================================================

  String? _actividadObservada;
  String? _actividadActual;
  String? _vehiculosMaquinaria;
  String? _cantidadArboles;

  final TextEditingController _descripcionController =
      TextEditingController();

  // ============================================================
  // EVIDENCIAS
  // ============================================================

  bool _tieneFoto = false;
  bool _tieneVideo = false;
  bool _tieneAudio = false;

  // ============================================================
  // UBICACIÓN
  // ============================================================

  bool _ubicacionConfirmada = false;
  LatLng? _coordenadasConfirmadas;

  // ============================================================
  // CONTROLADOR DE PÁGINAS
  // ============================================================

  final PageController _pageController = PageController();

  @override
  void dispose() {
    _descripcionController.dispose();
    _pageController.dispose();
    super.dispose();
  }

  // ============================================================
  // AVANZAR
  // ============================================================

  void _siguientePaso() {
    if (_currentStep == 0) {
      if (!_validarInformacion()) {
        return;
      }
    }

    if (_currentStep == 1) {
      if (!_validarEvidencia()) {
        return;
      }
    }

    if (_currentStep == 2) {
      if (!_ubicacionConfirmada) {
        _mostrarMensaje(
          'Confirma la ubicación del reporte para continuar.',
        );
        return;
      }
    }

    if (_currentStep < 3) {
      setState(() {
        _currentStep++;
      });

      _pageController.animateToPage(
        _currentStep,
        duration: const Duration(milliseconds: 300),
        curve: Curves.easeInOut,
      );
    }
  }

  // ============================================================
  // REGRESAR
  // ============================================================

  void _pasoAnterior() {
    if (_currentStep > 0) {
      setState(() {
        _currentStep--;
      });

      _pageController.animateToPage(
        _currentStep,
        duration: const Duration(milliseconds: 300),
        curve: Curves.easeInOut,
      );
    }
  }

  // ============================================================
  // VALIDAR INFORMACIÓN
  // ============================================================

  bool _validarInformacion() {
    if (_actividadObservada == null) {
      _mostrarMensaje(
        'Selecciona qué actividad observas.',
      );
      return false;
    }

    if (_actividadActual == null) {
      _mostrarMensaje(
        'Indica si la actividad está ocurriendo actualmente.',
      );
      return false;
    }

    if (_vehiculosMaquinaria == null) {
      _mostrarMensaje(
        'Indica si observas vehículos o maquinaria.',
      );
      return false;
    }

    if (_cantidadArboles == null) {
      _mostrarMensaje(
        'Selecciona una cantidad aproximada de árboles.',
      );
      return false;
    }

    return true;
  }

  // ============================================================
  // VALIDAR EVIDENCIA
  // ============================================================

  bool _validarEvidencia() {
    if (!_tieneFoto && !_tieneVideo && !_tieneAudio) {
      _mostrarMensaje(
        'Agrega al menos una evidencia: foto, video o audio.',
      );
      return false;
    }

    return true;
  }

  // ============================================================
  // MOSTRAR MENSAJE

    ScaffoldMessenger.of(context).hideCurrentSnackBar();

      SnackBar(
        content: Text(mensaje),
        behavior: SnackBarBehavior.floating,
      ),
    );
  }
